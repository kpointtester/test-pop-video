Reels — Lightweight Player VX

Reels rendered directly in the page flow (mode: "page") with lightWeightPlayer: true.

▶

View embed code 
<div id="vxplayer-reels-page"></div>
<script src="https://assets.zencite.in/orca/media/embed/player-vx.js"></script>
<script>
  window.VXPlayer(document.querySelector("#vxplayer-reels-page"), {
    hostname: "showcase-qa.zencite.in",
    type: "reels",
    mode: "page",
    videoIds: [
      "gcc-9c8393a9-f6c8-4d7f-98c0-03b8e60bfda7",
      "gcc-1d264a76-8fb7-4d63-bbe1-7646833029a0",
      "gcc-800e5504-848a-4571-bdb0-6b988f154b9f"
    ],
    autoScroll: true,
    loop: true,
    fullscreen: true,
    lightWeightPlayer: true
  }, { fab: { close: true } });
</script>
